var SHEET_NAME = "工作表1"; // 若頁籤有名稱請同步修改（如 "2627學年功輔留位記錄"） function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME) || ss.getActiveSheet(); var data = JSON.parse(e.postData.contents); var rows = sheet.getDataRange().getValues(); var rowIndex = -1; // 搜尋是否已有相同 ID 或 學生中文姓名 的記錄 for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(data.id) || (rows[i][1] === data.studentName && data.studentName !== "")) { rowIndex = i + 1; // 找到對應的列數 break; } } var rowData = [ data.id, data.studentName, data.targetGrade, data.parentPhone || '', data.daysPerWeek, data.duration, JSON.stringify(data.schedule), data.remarks, data.status || 'Pending', data.submittedAt ]; if (rowIndex > 0) { // 若已存在舊記錄，則直接覆蓋更新該列 sheet.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]); } else { // 若為新提交，則新增一列 sheet.appendRow(rowData); } return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } function doGet() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME) || ss.getActiveSheet(); var rows = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < rows.length; i++) { if (rows[i][0] === "") continue; // 跳過空列 result.push({ id: rows[i][0], studentName: rows[i][1], targetGrade: rows[i][2], parentPhone: rows[i][3], daysPerWeek: rows[i][4], duration: rows[i][5], schedule: typeof rows[i][6] === 'string' ? JSON.parse(rows[i][6] || '{}') : rows[i][6], remarks: rows[i][7], status: rows[i][8] || 'Pending', submittedAt: rows[i][9] }); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); }
