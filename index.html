var SHEET_NAME = "工作表1"; // 若頁籤名稱不同請同步修改 function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME) || ss.getActiveSheet(); var data = {}; // 解析 JSON 或 Form 表單資料 if (e.parameter && e.parameter.payload) { data = JSON.parse(e.parameter.payload); } else if (e.postData && e.postData.contents) { data = JSON.parse(e.postData.contents); } else if (e.parameter) { data = e.parameter; } var rows = sheet.getDataRange().getValues(); // 1. 執行真正刪除：比對 ID 並刪除整列 if (data.action === 'delete') { var targetId = String(data.id); for (var i = 1; i < rows.length; i++) { if (String(rows[i][0]) === targetId) { sheet.deleteRow(i + 1); // 刪除對應的 Google Sheet 列 return ContentService.createTextOutput(JSON.stringify({"result": "deleted"})).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({"result": "not_found"})).setMimeType(ContentService.MimeType.JSON); } // 防護機制：無姓名且無 ID 的無效請求直接拒絕，絕不新增空列 if (!data.studentName && !data.id) { return ContentService.createTextOutput(JSON.stringify({"result": "ignored_empty"})).setMimeType(ContentService.MimeType.JSON); } // 2. 執行覆蓋更新：若找到相同 ID 則覆蓋寫入，否則新增 var targetRowIndex = -1; for (var j = 1; j < rows.length; j++) { if (String(rows[j][0]) === String(data.id)) { targetRowIndex = j + 1; break; } } var rowData = [ data.id, data.studentName, data.targetGrade, data.parentPhone || '', data.daysPerWeek, data.duration, typeof data.schedule === 'object' ? JSON.stringify(data.schedule) : (data.schedule || '{}'), data.remarks || '無', data.status || 'Pending', data.submittedAt ]; if (targetRowIndex > 0) { sheet.getRange(targetRowIndex, 1, 1, rowData.length).setValues([rowData]); } else { sheet.appendRow(rowData); } return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"error": err.toString()})).setMimeType(ContentService.MimeType.JSON); } } function doGet() { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(SHEET_NAME) || ss.getActiveSheet(); var rows = sheet.getDataRange().getValues(); var result = []; for (var i = 1; i < rows.length; i++) { // 自動過濾 ID 為空或沒有姓名的資料 if (!rows[i][0] || !rows[i][1]) continue; result.push({ id: rows[i][0], studentName: rows[i][1], targetGrade: rows[i][2], parentPhone: rows[i][3], daysPerWeek: rows[i][4], duration: rows[i][5], schedule: typeof rows[i][6] === 'string' ? JSON.parse(rows[i][6] || '{}') : rows[i][6], remarks: rows[i][7], status: rows[i][8] || 'Pending', submittedAt: rows[i][9] }); } return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"error": err.toString()})).setMimeType(ContentService.MimeType.JSON); } }
